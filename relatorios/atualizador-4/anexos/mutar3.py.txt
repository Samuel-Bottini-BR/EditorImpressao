def v14(texto: str) -> str:
    """V14: recoloca o baixar antes da conferencia do motor."""
    t = texto.replace('    registrar(f"versão {alvo.Version} publicada e assinada pelo Samuel: baixando")\n'
                      "    gerente.download_updates(info)\n",
                      '    registrar(f"versão {alvo.Version} publicada e assinada pelo Samuel: baixando")\n', 1)
    return t.replace("    if not (motor_no_lugar or motor_esta_no_lugar)():\n",
                     "    gerente.download_updates(info)\n    if not (motor_no_lugar or motor_esta_no_lugar)():\n", 1)


linhas = []


def dizer(texto: str) -> None:
    """Escreve na tela e no arquivo de saida."""
    print(texto, flush=True)
    linhas.append(texto)
    if SAIDA:
        SAIDA.write_text("\n".join(linhas) + "\n", encoding="utf-8")


for nome, arq, velho, novo in M:
    if SO and nome.split()[0] not in SO:
        continue
    p = COPIA / arq
    orig = p.read_text(encoding="utf-8")
    if velho == "<V14>":
        mutado = v14(orig)
        if mutado.count("gerente.download_updates(info)") != 1 or mutado == orig:
            dizer(f"{nome}: TEXTO NAO ACHADO")
            continue
    else:
        if orig.count(velho) != 1:
            dizer(f"{nome}: TEXTO NAO ACHADO ({orig.count(velho)})")
            continue
        mutado = orig.replace(velho, novo)
    p.write_text(mutado, encoding="utf-8")
    pegos = []
    testes = TESTES + (["test_windows_peca.py"] if arq == W else []) + (["test_ocr_kraken.py"] if arq == K else [])
    try:
        for t in testes:
            r = subprocess.run([PY, "-m", "pytest", f"tests/{t}", "-q", "-p", "no:cacheprovider", "-W", "ignore",
                                "--tb=no", "-rf"], cwd=COPIA, capture_output=True, text=True, encoding="utf-8",
                               errors="replace", timeout=900)
            if r.returncode != 0:
                falhou = [l.split(" - ")[0] for l in r.stdout.splitlines() if l.startswith("FAILED")]
                ultima = (r.stdout.strip().splitlines() or ["?"])[-1]
                pegos.append(f"{t}: {ultima} {falhou[:3]}")
    finally:
        p.write_text(orig, encoding="utf-8")
    dizer(f"{nome}: {'ACUSADA' if pegos else 'NAO ACUSADA'}")
    for x in pegos:
        dizer("    " + x[:400])
