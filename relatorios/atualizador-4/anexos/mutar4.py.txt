def rodar(teste: str, filtro: str) -> tuple[bool, str]:
    """Roda o arquivo de teste com o filtro; True se algum teste falhou (a mutacao foi acusada)."""
    feito = subprocess.run([PY, "-m", "pytest", teste, "-q", "-p", "no:cacheprovider", "-k", filtro], cwd=RAIZ,
                           capture_output=True, text=True, timeout=600)
    ultima = (feito.stdout.strip().splitlines() or ["?"])[-1]
    return feito.returncode != 0 and "failed" in ultima, ultima


linhas = []
acusadas = 0
for nome, arquivo, hoje, desligado, teste, filtro in MUTACOES:
    caminho = RAIZ / arquivo
    original = caminho.read_bytes()
    texto = original.decode("utf-8")
    if "\r\n" in texto:                       # a copia de trabalho no Windows pode estar com CRLF (core.autocrlf)
        hoje, desligado = hoje.replace("\n", "\r\n"), desligado.replace("\n", "\r\n")
    if texto.count(hoje) != 1:
        linhas.append(f"{nome}: TEXTO NAO ACHADO UMA VEZ ({texto.count(hoje)})")
        continue
    try:
        caminho.write_text(texto.replace(hoje, desligado), encoding="utf-8", newline="")
        acusada, ultima = rodar(teste, filtro)
    finally:
        caminho.write_bytes(original)
    acusadas += acusada
    linhas.append(f"{nome}: {'ACUSADA' if acusada else 'NAO ACUSADA'} ({teste} -k \"{filtro}\": {ultima})")
    print(linhas[-1], flush=True)
linhas.append(f"{acusadas} de {len(MUTACOES)} acusadas")
(RAIZ / "relatorios/atualizador-4/anexos/mutacoes-volta4.txt").write_text("\n".join(linhas) + "\n", encoding="utf-8")
print(linhas[-1])
