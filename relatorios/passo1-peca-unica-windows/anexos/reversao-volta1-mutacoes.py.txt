def rodar(teste: str) -> bool:
    """Roda um teste na cópia; True = passou."""
    ambiente = {**os.environ, "QT_QPA_PLATFORM": "offscreen", "PYTHONIOENCODING": "utf-8"}
    feito = subprocess.run([PY, "-m", "pytest", teste, "-q", "-p", "no:cacheprovider"], cwd=COPIA,
                           capture_output=True, text=True, env=ambiente, timeout=600)
    return feito.returncode == 0


def main() -> int:
    """Roda o teste de cada mutação: primeiro sem mutar (tem de passar), depois mutado (tem de falhar)."""
    resultado = []
    for nome, arquivo, velho, novo, teste in MUTACOES:
        caminho = COPIA / arquivo
        original = caminho.read_text(encoding="utf-8")
        if original.count(velho) != 1:
            resultado.append(f"NÃO APLICADA ({original.count(velho)}x): {nome}")
            continue
        limpo = rodar(teste)
        caminho.write_text(original.replace(velho, novo), encoding="utf-8")
        try:
            mutado = rodar(teste)
        finally:
            caminho.write_text(original, encoding="utf-8")
        estado = "ACUSADA" if limpo and not mutado else ("o teste não passa sem mutar" if not limpo else "NÃO ACUSADA")
        resultado.append(f"{estado}: {nome}  [{teste.split('::')[1]}]")
        print(resultado[-1], flush=True)
    print("\n".join(["", "RESUMO:"] + resultado))
    return 0 if all(r.startswith("ACUSADA") for r in resultado) else 1


if __name__ == "__main__":
    sys.exit(main())
