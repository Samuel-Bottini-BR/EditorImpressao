ambiente = {**os.environ, "QT_QPA_PLATFORM": "offscreen"}
for codigo, o_que, arquivo, velho, novo, testes in MUTACOES:
    caminho = copia / arquivo
    original = caminho.read_bytes()
    texto = original.decode("utf-8")
    if "\r\n" in texto:  # o git archive pode entregar com CRLF
        velho, novo = velho.replace("\n", "\r\n"), novo.replace("\n", "\r\n")
    if SO and codigo not in SO:
        continue
    if texto.count(velho) != 1:
        print(f"{codigo}: NÃO APLICADA (o texto aparece {texto.count(velho)} vezes) - {o_que}", flush=True)
        continue
    caminho.write_bytes(texto.replace(velho, novo).encode("utf-8"))
    try:
        for teste in testes:
            r = subprocess.run([PY, "-m", "pytest", teste, "-q", "-p", "no:cacheprovider", "-x"], cwd=copia,
                               capture_output=True, text=True, env=ambiente, timeout=600)
            ultima = (r.stdout.strip().splitlines() or ["?"])[-1]
            print(f"{codigo}: {'ACUSOU' if r.returncode != 0 else 'NÃO ACUSOU'} - {o_que} - {teste.split('::')[-1]}"
                  f" ({ultima})", flush=True)
    finally:
        caminho.write_bytes(original)
