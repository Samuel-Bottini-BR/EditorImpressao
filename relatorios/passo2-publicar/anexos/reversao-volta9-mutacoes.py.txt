def rodar_testes(arquivo: str) -> tuple[int, list[str]]:
    """Roda o arquivo de teste na copia; devolve (codigo, os que falharam e a linha do fim)."""
    r = subprocess.run([PY, "-m", "pytest", arquivo, "-q", "-p", "no:cacheprovider", "-rf"], cwd=COPIA,
                       capture_output=True, text=True, encoding="utf-8", errors="replace",
                       env={**os.environ, "QT_QPA_PLATFORM": "offscreen"},
                       creationflags=getattr(subprocess, "CREATE_NO_WINDOW", 0))
    falhas = [l.split(" - ")[0].replace("FAILED ", "") for l in r.stdout.splitlines() if l.startswith("FAILED")]
    ultima = [l for l in r.stdout.splitlines() if " passed" in l or " failed" in l]
    return r.returncode, falhas + ultima[-1:]


linhas = []
for codigo, o_que, arquivo, velho, novo, teste in MUTACOES:
    caminho = COPIA / arquivo
    original = caminho.read_bytes()
    texto = original.decode("utf-8")
    if "\r\n" in texto:
        velho, novo = velho.replace("\n", "\r\n"), novo.replace("\n", "\r\n")
    if texto.count(velho) != 1:
        linhas.append(f"{codigo}: NAO APLICADA (o trecho aparece {texto.count(velho)} vezes): {o_que}")
        continue
    caminho.write_bytes(texto.replace(velho, novo).encode("utf-8"))
    try:
        rc, falhas = rodar_testes(teste)
    finally:
        caminho.write_bytes(original)
    assert caminho.read_bytes() == original
    linhas.append(f"{codigo} ({o_que}; {arquivo}; {teste}): {'ACUSADA' if rc != 0 else 'NAO ACUSADA'}")
    linhas += [f"    {f}" for f in falhas]
    SAIDA.write_text("\n".join(linhas) + "\n", encoding="utf-8")
for teste in (PASTA, "tests/test_empacotar.py"):
    rc, falhas = rodar_testes(teste)
    linhas.append(f"sem mutacao ({teste}): codigo {rc}; {falhas[-1] if falhas else ''}")
SAIDA.write_text("\n".join(linhas) + "\n", encoding="utf-8")
print("\n".join(linhas))
