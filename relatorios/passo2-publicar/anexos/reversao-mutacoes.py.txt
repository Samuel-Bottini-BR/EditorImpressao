resultado = []
for nome, arquivo, velho, novo, teste in MUTACOES:
    copia = LIMPA.parent / "mut"
    shutil.rmtree(copia, ignore_errors=True)
    shutil.copytree(LIMPA, copia)
    alvo = copia / arquivo
    texto = alvo.read_text(encoding="utf-8")
    if texto.count(velho) != 1:
        resultado.append(f"{nome}: NÃO APLICADA (o texto não está uma vez só)")
        continue
    alvo.write_text(texto.replace(velho, novo), encoding="utf-8")
    r = subprocess.run([PY, "-m", "pytest", teste, "-q", "-p", "no:cacheprovider", "-x"], cwd=copia,
                       capture_output=True, text=True, encoding="utf-8", errors="replace",
                       env={**__import__("os").environ, "QT_QPA_PLATFORM": "offscreen"})
    ultima = (r.stdout.strip().splitlines() or ["?"])[-1]
    resultado.append(f"{nome}: {'ACUSADA' if r.returncode != 0 else 'PASSOU (não acusada)'} ({ultima}) [{teste}]")
limpo = subprocess.run([PY, "-m", "pytest", "tests/test_publicar_versao.py", "tests/test_publicar_motor.py",
                        "tests/test_publicar_servidor.py", "-q", "-p", "no:cacheprovider"], cwd=LIMPA,
                       capture_output=True, text=True, encoding="utf-8", errors="replace",
                       env={**__import__("os").environ, "QT_QPA_PLATFORM": "offscreen"})
resultado.append(f"cópia limpa: {(limpo.stdout.strip().splitlines() or ['?'])[-1]}")
print("\n".join(resultado))
