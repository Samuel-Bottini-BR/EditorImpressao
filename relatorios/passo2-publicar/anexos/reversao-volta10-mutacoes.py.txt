def main() -> None:
    linhas = []
    for nome, arquivo, o_que, velho, novo, teste, filtro in MUTACOES:
        caminho = COPIA / arquivo
        original = caminho.read_bytes()
        texto = original.decode("utf-8")
        texto_lf = texto.replace("\r\n", "\n")
        assert texto_lf.count(velho) == 1, (nome, "o trecho não bate")
        mudado = texto_lf.replace(velho, novo)
        if "\r\n" in texto:
            mudado = mudado.replace("\n", "\r\n")
        caminho.write_bytes(mudado.encode("utf-8"))
        inicio = time.perf_counter()
        try:
            r = subprocess.run([PY, "-m", "pytest", teste, "-q", "-p", "no:cacheprovider", "-k", filtro], cwd=COPIA,
                               capture_output=True, text=True, encoding="utf-8", errors="replace", timeout=1200)
            fim = (r.stdout.strip().splitlines() or ["(sem saída)"])[-1]
            acusada = r.returncode != 0 and "failed" in fim
        finally:
            caminho.write_bytes(original)
        assert caminho.read_bytes() == original, (nome, "não voltou ao original")
        linha = (f"{nome} | {arquivo} | {o_que} | {teste} -k \"{filtro}\" | {fim} | "
                 f"{'ACUSADA' if acusada else 'NAO ACUSADA'} | {time.perf_counter() - inicio:.0f} s | devolvido igual")
        print(linha, flush=True)
        linhas.append(linha)
    SAIDA.write_text("\n".join(linhas) + "\n", encoding="utf-8")


if __name__ == "__main__":
    main()
