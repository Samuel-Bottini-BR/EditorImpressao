def main() -> int:
    ambiente = dict(os.environ, QT_QPA_PLATFORM="offscreen", PYTHONIOENCODING="utf-8")
    acusadas = 0
    for codigo, o_que, arquivo, velho, novo, teste, filtro in MUTACOES:
        alvo = COPIA / arquivo
        original = alvo.read_bytes()
        texto = original.decode("utf-8")
        if texto.count(velho) != 1:
            print(f"{codigo}: o trecho não foi achado uma vez só ({texto.count(velho)}): NÃO RODADA")
            continue
        alvo.write_bytes(texto.replace(velho, novo).encode("utf-8"))
        try:
            r = subprocess.run([PY, "-m", "pytest", teste, "-q", "-k", filtro, "-p", "no:cacheprovider", "--tb=no"],
                               cwd=COPIA, env=ambiente, capture_output=True, text=True, encoding="utf-8",
                               errors="replace", timeout=900)
        finally:
            alvo.write_bytes(original)
        assert alvo.read_bytes() == original
        fim = [linha for linha in r.stdout.splitlines() if "passed" in linha or "failed" in linha][-1:]
        acusou = r.returncode != 0
        acusadas += acusou
        print(f"{codigo} ({o_que}) -> {teste} -k {filtro}: {'ACUSADA' if acusou else 'NAO acusada'} {fim}")
    print(f"\n{acusadas} de {len(MUTACOES)} acusadas; cada arquivo devolvido ao original (conferido byte a byte).")
    return 0


if __name__ == "__main__":
    sys.exit(main())
