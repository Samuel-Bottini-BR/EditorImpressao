def rodar(teste: str, filtro: str) -> str:
    """Roda um arquivo de teste com o filtro; devolve a ultima linha do pytest."""
    r = subprocess.run([PYTHON, "-m", "pytest", teste, "-q", "-p", "no:cacheprovider", "-k", filtro],
                       cwd=COPIA, capture_output=True, text=True, encoding="utf-8", errors="replace", timeout=1800)
    linhas = [l for l in r.stdout.splitlines() if l.strip()]
    falhas = [l for l in linhas if l.startswith(("FAILED", "ERROR"))]
    return (linhas[-1] if linhas else r.stderr[-300:]) + "".join(f"\n      {f}" for f in falhas)


saida = []
for nome, arquivo, velho, novo, testes in MUTACOES:
    caminho = COPIA / arquivo
    original = caminho.read_bytes()
    texto = original.decode("utf-8")
    fim = "\r\n" if "\r\n" in texto else "\n"
    velho_aqui, novo_aqui = velho.replace("\n", fim), novo.replace("\n", fim)
    assert texto.count(velho_aqui) == 1, (nome, texto.count(velho_aqui))
    caminho.write_bytes(texto.replace(velho_aqui, novo_aqui).encode("utf-8"))
    try:
        saida.append(nome)
        for teste, filtro in testes:
            saida.append(f"  {teste} -k \"{filtro}\": {rodar(teste, filtro)}")
    finally:
        caminho.write_bytes(original)
    assert caminho.read_bytes() == original
    print(saida[-len(testes) - 1:], flush=True)
saida.append("sem mutacao (a copia como esta):")

RESULTADO.write_text("\n".join(saida) + "\n", encoding="utf-8")
