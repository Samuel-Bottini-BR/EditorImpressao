def rodar_testes() -> tuple[int, list[str]]:
    """Roda o arquivo de testes do publicar inteiro na copia; devolve (codigo, os que falharam)."""
    r = subprocess.run([PY, "-m", "pytest", "tests/test_publicar_versao.py", "-q", "-p", "no:cacheprovider", "-rf"],
                       cwd=COPIA, capture_output=True, text=True, encoding="utf-8", errors="replace",
                       env={**__import__("os").environ, "QT_QPA_PLATFORM": "offscreen"})
    falhas = [l.split(" - ")[0].replace("FAILED ", "") for l in r.stdout.splitlines() if l.startswith("FAILED")]
    ultima = [l for l in r.stdout.splitlines() if " passed" in l or " failed" in l]
    return r.returncode, falhas + ultima[-1:]


SO = set(sys.argv[3].split(",")) if len(sys.argv) > 3 else None   # rodar so estas (ex.: U2,U5,U12)
linhas = []
for codigo, o_que, arquivo, velho, novo in MUTACOES:
    if SO is not None and codigo not in SO:
        continue
    caminho = COPIA / arquivo
    original = caminho.read_bytes()
    texto = original.decode("utf-8")
    if "\r\n" in texto:          # a copia do git archive tem quebra de linha do Windows
        velho, novo = velho.replace("\n", "\r\n"), novo.replace("\n", "\r\n")
    if texto.count(velho) != 1:
        linhas.append(f"{codigo}: NAO APLICADA (o trecho aparece {texto.count(velho)} vezes): {o_que}")
        continue
    caminho.write_bytes(texto.replace(velho, novo).encode("utf-8"))
    try:
        rc, falhas = rodar_testes()
    finally:
        caminho.write_bytes(original)
    assert caminho.read_bytes() == original
    acusada = "ACUSADA" if rc != 0 else "NAO ACUSADA"
    linhas.append(f"{codigo} ({o_que}; {arquivo}): {acusada}")
    linhas += [f"    {f}" for f in falhas]
    SAIDA.write_text("\n".join(linhas) + "\n", encoding="utf-8")
rc, falhas = rodar_testes()
linhas.append(f"sem mutacao (a copia devolvida ao original): codigo {rc}; {falhas[-1] if falhas else ''}")
SAIDA.write_text("\n".join(linhas) + "\n", encoding="utf-8")
print("\n".join(linhas))
