def rodar(testes: list[str]) -> tuple[bool, str]:
    """Roda os testes na cópia; (todos passaram?, a última linha)."""
    ambiente = {**os.environ, "QT_QPA_PLATFORM": "offscreen"}
    feito = subprocess.run([PY, "-m", "pytest", "-q", "-p", "no:cacheprovider", *testes], cwd=COPIA,
                           capture_output=True, text=True, encoding="utf-8", errors="replace", env=ambiente)
    linhas = [l for l in feito.stdout.splitlines() if l.strip()]
    return feito.returncode == 0, (linhas[-1] if linhas else feito.stderr[-200:])

        caminho.write_text(atual.replace(velho, novo), encoding="utf-8")
        ok, fim = rodar(testes)
        linhas.append(f"{nome}: {'NÃO acusada' if ok else 'acusada'} ({fim})")
    finally:
        for arq, texto in guardados.items():
            (COPIA / arq).write_text(texto, encoding="utf-8")

SAIDA.write_text("\n".join(linhas) + "\n", encoding="utf-8")
