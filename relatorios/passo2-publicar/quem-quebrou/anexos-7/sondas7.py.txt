def _pronta_com(a, nome: str, conteudo: bytes, versao_txt: bytes | None = b"14.44.35211.0") -> Path:
    pronta = _pasta_pronta(a["tmp"] / nome / "EditorImpressao", versao.VERSAO)

@pytest.mark.parametrize("versao_txt", [b"\xef\xbb\xbf14.44.35211.0", b"14.44.35211.0\r\n", b"\xff\xfe1\x004\x00",
                                        b"", b" 14.44.35211.0 "])
def test_sonda7_3_o_arquivo_da_versao_lido_igual_pelo_programa_instalado(ambiente, versao_txt):
    """A conferência e o programa instalado (atualizacao.instalar_visual_c_se_faltar) leem o mesmo arquivo com código
    diferente: o que passa na conferência tem de ser lido certo pelo programa."""

    r = empacotar.conferir_pasta_pronta(pronta, versao.VERSAO)
    rodou = []
    try:

                                                     rodar=lambda c, t: (rodou.append(c), (0, ""))[1])

    except Exception as erro:  # noqa: BLE001
        programa = f"LEVANTOU {erro!r}"
    print("SONDA7-3", repr(versao_txt), "conferencia:", r or "PASSA", "| programa instalado (sem VC no PC):", programa)


@pytest.mark.parametrize("resposta", ["falha", "vazio", "chaves", "so-status"])
def test_sonda7_4_a_leitura_que_falha_nunca_vira_passou(ambiente, monkeypatch, resposta):
    from core import windows
    pronta = _pronta_com(ambiente, "l-" + resposta, (_PECAS7 / "oficial.exe").read_bytes())

    def powershell(texto, limite_s=600, segredo=None):
        if resposta == "falha":
            return windows._falhou("o PowerShell não terminou no tempo", "limite")
        saida = {"vazio": "", "chaves": "{}", "so-status": '{"status":"Valid"}'}[resposta]
        return windows._certo(saida=saida, erros="")

    monkeypatch.setattr(windows, "rodar_powershell", powershell)
    r = empacotar.conferir_pasta_pronta(pronta, versao.VERSAO)
    monkeypatch.setattr(empacotar, "DOWNLOADS", _PECAS7 / "so-oficial")
    (_PECAS7 / "so-oficial").mkdir(exist_ok=True)
    alvo = _PECAS7 / "so-oficial" / empacotar.VC_REDIST_NOME
    if not alvo.exists():
        shutil.copyfile(_PECAS7 / "oficial.exe", alvo)
    m = empacotar.preparar_vc_redist(baixar=False)
    print("SONDA7-4", resposta, "| pasta:", r, "| montagem:", m[0], repr(m[1]), m[2])
    assert r and m[0] is None
