def rodar(testes):
    amb = {**os.environ, "QT_QPA_PLATFORM": "offscreen"}
    r = subprocess.run([PY, "-m", "pytest", *testes, "-q", "-p", "no:cacheprovider", "-x"], cwd=COPIA,
                       capture_output=True, text=True, encoding="utf-8", errors="replace", env=amb, timeout=1800)
    ultima = [l for l in r.stdout.splitlines() if "passed" in l or "failed" in l or "error" in l.lower()]
    falhou = [l for l in r.stdout.splitlines() if l.startswith("FAILED") or l.startswith("ERROR")]
    return r.returncode, (ultima[-1] if ultima else r.stdout[-300:]), falhou[:3]


linhas = []
for nome, arquivo, de, para, testes in M:
    caminho = COPIA / arquivo
    original = caminho.read_text(encoding="utf-8")
    if original.count(de) != 1:
        linhas.append(f"{nome}: NAO APLICADA (o trecho aparece {original.count(de)} vezes)")
        continue
    caminho.write_text(original.replace(de, para), encoding="utf-8")
    try:
        codigo, resumo, falhou = rodar(testes)
    finally:
        caminho.write_text(original, encoding="utf-8")
    linhas.append(f"{nome}: {'ACUSADA' if codigo != 0 else 'NAO ACUSADA'} ({resumo}) {falhou}")
    print(linhas[-1], flush=True)
SAIDA.write_text("\n".join(linhas) + "\n", encoding="utf-8")
