for nome, arquivo, velho, novo, teste in MUTACOES:
    if so and not any(nome.startswith(s) for s in so):
        continue
    alvo = COPIA / arquivo
    original = alvo.read_text(encoding="utf-8")
    n = original.count(velho)
    if n != 1:
        print(f"{nome}: o pedaco aparece {n} vez(es); NAO RODADA", flush=True)
        continue
    alvo.write_text(original.replace(velho, novo), encoding="utf-8")
    try:
        r = subprocess.run([PY, "-m", "pytest", *teste, "-q", "-p", "no:cacheprovider", "-x"], cwd=COPIA,
                           capture_output=True, text=True, encoding="utf-8", errors="replace", timeout=1500)
        linhas = [l for l in r.stdout.splitlines() if l.strip()]
        falhas = [l for l in linhas if l.startswith("FAILED")]
        print(f"{nome}: {linhas[-1] if linhas else r.stderr[-300:]}", flush=True)
        for f in falhas[:4]:
            print(f"    {f[:200]}", flush=True)
    finally:
        alvo.write_text(original, encoding="utf-8")
print("sem mutacao: conferir a copia volta igual ->",
      subprocess.run(["git", "--no-pager", "diff", "--no-index", "--stat", str(COPIA / "publicar_versao_kaique.py"),
                      str(COPIA / "publicar_versao_kaique.py")], capture_output=True, text=True).returncode)
