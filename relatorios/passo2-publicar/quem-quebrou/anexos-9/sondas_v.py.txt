elif qual == "s2":
    import ctypes
    padrao = Path(r"D:\programas\EditorImpressao-arquivos\ferramentas")
    buf = ctypes.create_unicode_buffer(600)
    ctypes.windll.kernel32.GetShortPathNameW(str(padrao), buf, 600)
    curto = buf.value
    jeitos = {
        "minusculas": r"d:\programas\editorimpressao-arquivos\ferramentas",
        "barras /": "D:/programas/EditorImpressao-arquivos/ferramentas",
        "barra no fim": "D:\\programas\\EditorImpressao-arquivos\\ferramentas\\",

        print(f"{nome:18} | igual ao de fabrica (o que os scripts comparam): {p == padrao!s:5} | "

elif qual == "s3":
    novo = Path(sys.argv[2])
    sys.path.insert(0, str(novo))
    regra = novo / "core" / "pasta_das_ferramentas.py"
    original = regra.read_bytes()
    linha = 'PASTA_DAS_FERRAMENTAS = Path(r"D:\\programas\\EditorImpressao-arquivos\\ferramentas")'
    # o "seguro mexer" do docstring: a linha muda de jeito (um comentario no fim, ou a pasta muda de lugar)
    jeitos = {
        "comentario no fim": linha + "  # D-820",
        "a pasta mudou de lugar": linha.replace("D:\\programas", "E:\\programas"),
    }
    import importlib.util
    spec = importlib.util.spec_from_file_location("t", novo / "tests" / "test_pasta_das_ferramentas.py")
    try:
        for nome, nova_linha in jeitos.items():
            regra.write_bytes(original.decode("utf-8").replace(linha, nova_linha).encode("utf-8"))
            t = importlib.util.module_from_spec(spec)
            spec.loader.exec_module(t)
            destino = Path(__file__).resolve().parent / f"s3-{nome.replace(' ', '-')}"
            destino.mkdir(exist_ok=True)
            t._copiar_o_codigo(destino, Path(r"C:\pasta de prova\ferramentas"))
            copiado = (destino / "core" / "pasta_das_ferramentas.py").read_text(encoding="utf-8")
            achada = [l for l in copiado.splitlines() if l.startswith("PASTA_DAS_FERRAMENTAS")]
            print(f"{nome}: a copia do teste ficou com: {achada}")
    finally:
        regra.write_bytes(original)
    print("devolvido igual:", regra.read_bytes() == original)
