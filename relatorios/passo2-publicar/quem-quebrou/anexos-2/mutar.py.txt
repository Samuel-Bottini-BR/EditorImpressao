def rodar(copia: Path, nome: str) -> str:
    arquivo, velho, novo, teste, filtro = M[nome]
    caminho = copia / arquivo
    original = caminho.read_bytes()
    texto = original.decode("utf-8")
    if '\r\n' in texto:
        velho, novo = velho.replace('\n', '\r\n'), novo.replace('\n', '\r\n')
    n = texto.count(velho)
    if n != 1:
        return f"{nome}: TRECHO ACHADO {n} VEZES (mutacao nao aplicada)"
    caminho.write_bytes(texto.replace(velho, novo).encode("utf-8"))
    try:
        cmd = [PY, "-m", "pytest", teste, "-q", "-p", "no:cacheprovider", "-x"]
        if filtro:
            cmd += ["-k", filtro]
        r = subprocess.run(cmd, cwd=copia, capture_output=True, text=True, encoding="utf-8", errors="replace",
                           env={**__import__("os").environ, "QT_QPA_PLATFORM": "offscreen"}, timeout=900)
        fim = [l for l in r.stdout.splitlines() if l.strip()][-1:]
        falhou = [l for l in r.stdout.splitlines() if l.startswith("FAILED")]
        veredito = "ACUSADA" if r.returncode != 0 else "PASSOU (não acusada)"
        return f"{nome}: {veredito} | {fim[0] if fim else ''} | {'; '.join(falhou)[:300]}"
    finally:
        caminho.write_bytes(original)


if __name__ == "__main__":
    copia = Path(sys.argv[1])
    for nome in (sys.argv[2:] or list(M)):
        print(rodar(copia, nome), flush=True)
