def rodar(testes):
    r = subprocess.run([PY, "-B", "-m", "pytest", *testes, "-q", "-p", "no:cacheprovider", "-x", "--no-header", "-rf"],
                       cwd=COPIA, capture_output=True, text=True, encoding="utf-8", errors="replace",
                       env={**__import__("os").environ, "QT_QPA_PLATFORM": "offscreen"}, timeout=1800)
    linhas = [l for l in r.stdout.splitlines() if l.startswith("FAILED") or l.startswith("ERROR")
              or " passed" in l or " failed" in l or " error" in l]
    return r.returncode, linhas


def main(escolhidas):
    for codigo, o_que, arquivo, de, para, testes in M:
        if escolhidas and codigo not in escolhidas:
            continue
        caminho = COPIA / arquivo
        original = caminho.read_bytes()
        texto = original.decode("utf-8")
        if "\r\n" in texto:              # o git archive desta maquina grava com CRLF
            de, para = de.replace("\n", "\r\n"), para.replace("\n", "\r\n")
        if texto.count(de) != 1:
            print(f"{codigo}: o trecho aparece {texto.count(de)} vezes; mutacao nao aplicada", flush=True)
            continue
        try:
            caminho.write_bytes(texto.replace(de, para).encode("utf-8"))
            codigo_saida, linhas = rodar(testes)
        finally:
            caminho.write_bytes(original)
        assert caminho.read_bytes() == original
        veredito = "ACUSADA" if codigo_saida != 0 else "PASSOU (nenhum teste viu)"
        print(f"{codigo} ({o_que}; {arquivo}; {', '.join(testes)}): {veredito}", flush=True)
        for l in linhas:
            print(f"    {l}", flush=True)


if __name__ == "__main__":
    main(sys.argv[1:])
