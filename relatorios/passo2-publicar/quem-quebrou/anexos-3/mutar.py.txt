for codigo, oque, arq, de, para, testes in MUT:
    caminho = RAIZ / arq
    original = caminho.read_bytes()
    texto = original.decode("utf-8")
    n = texto.count(de)
    if n != 1:
        print(f"{codigo}: NAO APLICADA ({n} vezes) - {oque}", flush=True)
        continue
    caminho.write_bytes(texto.replace(de, para).encode("utf-8"))
    try:
        r = subprocess.run([PY, "-m", "pytest", *testes, "-q", "-x", "-p", "no:cacheprovider"], cwd=RAIZ,
                           capture_output=True, text=True, encoding="utf-8", errors="replace", timeout=900)

        falhou = [l for l in r.stdout.splitlines() if l.startswith("FAILED")]
        print(f"{codigo}: {'ACUSOU' if r.returncode else 'NAO ACUSOU'} - {oque} - {fim} {falhou[:1]}", flush=True)
    finally:
        caminho.write_bytes(original)
