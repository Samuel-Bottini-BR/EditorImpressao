COPIA = Path(r"D:/programas/EditorImpressao/.claude/worktrees/passo2")
teste = (COPIA / "tests/test_publicar_versao.py").read_text(encoding="utf-8")
arv = ast.parse(teste)
pedacos = [n for n in arv.body if (isinstance(n, ast.FunctionDef) and n.name == "_preposicao_a_mao")

assert len(pedacos) == 3, len(pedacos)
ns = {}
exec(compile(ast.Module(body=pedacos, type_ignores=[]), "guarda", "exec"), ns)

    ("H1", "liberada em a pasta (soma com um f-string que so tem o nome)", "erro", L836,

     "raise Parou(f\"o motor publicado em {com_preposicao('em', destino.nome)} não confere: {motivo}\")"),

     "print(f\"  {etiqueta} liberada em\" + f\" {destino.nome}\")"),

    ("F8a", "com_preposicao com nome do argumento (certo)", "certo", L836,

print("sem mutacao:", guarda(fonte))
for cod, oque, esperado, velho, novo in MUT:
    assert fonte.count(velho) == 1, (cod, fonte.count(velho))
    mut = fonte.replace(velho, novo)
    compile(mut, "m", "exec")
    achou = guarda(mut)
    veredito = ("ACUSADA" if achou else "PASSA") 
    bom = (achou and esperado == "erro") or (not achou and esperado == "certo")
    print(f"{cod} [{esperado}] {oque}: {veredito}{'' if bom else '  <-- FURO' if esperado=='erro' else '  <-- FALSO ALARME'}"
