def main() -> int:
    ambiente = dict(os.environ, QT_QPA_PLATFORM="offscreen", PYTHONIOENCODING="utf-8")
    acusadas = 0
    for codigo, o_que, arquivo, velho, novo, testes in MUTACOES:
        alvo = COPIA / arquivo
        original = alvo.read_bytes()
        texto = original.decode("utf-8")
        if texto.count(velho) != 1:
            print(f"{codigo}: o trecho nao foi achado uma vez so ({texto.count(velho)}): NAO RODADA", flush=True)
            continue
        alvo.write_bytes(texto.replace(velho, novo).encode("utf-8"))
        resultados = []
        try:
            for teste in testes:
                r = subprocess.run([PY, "-m", "pytest", teste, "-q", "-p", "no:cacheprovider", "--tb=no", "-rf"],
                                   cwd=COPIA, env=ambiente, capture_output=True, text=True, encoding="utf-8",
                                   errors="replace", timeout=1800)
                fim = [l for l in r.stdout.splitlines() if " passed" in l or " failed" in l or "error" in l][-1:]
                falhos = [l.split(" - ")[0] for l in r.stdout.splitlines() if l.startswith("FAILED")]
                resultados.append((teste, r.returncode, fim, falhos))
        finally:
            alvo.write_bytes(original)
        assert alvo.read_bytes() == original
        acusou = any(rc != 0 for _, rc, _, _ in resultados)
        acusadas += acusou
        print(f"{codigo} ({o_que}): {'ACUSADA' if acusou else 'NAO ACUSADA'}", flush=True)
        for teste, rc, fim, falhos in resultados:
            print(f"    {teste}: {fim}", flush=True)
            for f in falhos[:6]:
                print(f"        {f}", flush=True)
    print(f"\n{acusadas} de {len(MUTACOES)} acusadas; cada arquivo devolvido ao original (conferido byte a byte).")
    return 0


if __name__ == "__main__":
    sys.exit(main())
