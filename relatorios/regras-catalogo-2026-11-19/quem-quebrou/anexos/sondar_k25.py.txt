def rodar_py():
    print("=== Python ===")
    for rotulo, esperado, linha in PY:
        fonte = CAB + "    " + linha + "\n"
        try:
            achados = [a for a in porteiro.conferir_codigo(["scripts/sonda.py"], lambda _c, f=fonte: f) if a.conferencia == "K25"]
        except Exception as e:  # noqa: BLE001
            print(f"ERRO   {rotulo}: {e!r}")
            continue
        acusou = bool(achados)
        marca = "ok   " if acusou == esperado else ("PASSA" if esperado else "FALSO")
        print(f"{marca} acusou={acusou!s:5} {rotulo}")


def rodar_term():
    print("=== Terminal ===")
    for ext, rotulo, esperado, linha in TERM:
        nome = f"scripts/sonda{ext}"
        fonte = "#!/usr/bin/env bash\n" + linha + "\n"
        achados = [a for a in porteiro.conferir_codigo([nome], lambda _c, f=fonte: f) if a.conferencia == "K25"]
        acusou = bool(achados)
        marca = "ok   " if acusou == esperado else ("PASSA" if esperado else "FALSO")
        print(f"{marca} acusou={acusou!s:5} {ext:6} {rotulo}")


rodar_py()
rodar_term()
