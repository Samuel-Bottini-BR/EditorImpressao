"""Prova de pesquisa (NAO e codigo do programa): livreto A4 deitada com duas A5,
margens 1,3 cm (fora/cima/baixo) e 1,0 cm (costura), escala por lado, aviso de
margem. Usa a ordem do core/cadernos.py da worktree (so leitura).

Uso: python prova_livreto.py <pasta_worktree> <pasta_saida>
"""
import sys, time
from pathlib import Path

import fitz
import numpy as np
import cv2

WT, SAIDA = Path(sys.argv[1]), Path(sys.argv[2])
sys.path.insert(0, str(WT))
from core.cadernos import montar_ordem, conferir_sequencia  # noqa: E402

CM = 72 / 2.54
A4_DEITADA = fitz.Rect(0, 0, 29.7 * CM, 21.0 * CM)
FORA, COSTURA, IMPRESSORA = 1.3, 1.0, 0.3   # cm


def livro_falso(caminho, n, larg_cm=14.6, alt_cm=23.5, dpi=100, ruido=False, m_cm=0.4):
    """Paginas-imagem como as que o programa grava (uma imagem por pagina).
    Mancha de texto com margem de 1,5 cm do papel do livro; numero grande."""
    doc = fitz.open()
    w, h = int(larg_cm / 2.54 * dpi), int(alt_cm / 2.54 * dpi)
    m = int(m_cm / 2.54 * dpi)
    for i in range(n):
        img = np.full((h, w), 255, np.uint8)
        if ruido:  # imagem pouco compressivel, para medir memoria
            img[m:h - m, m:w - m] = np.random.randint(0, 255, (h - 2 * m, w - 2 * m), np.uint8)
        else:
            img[m:h - m, m:w - m] = 225
            cv2.putText(img, str(i + 1), (w // 2 - 60, h // 2), cv2.FONT_HERSHEY_SIMPLEX, 4, 0, 10)
        ok, png = cv2.imencode(".png", img)
        p = doc.new_page(width=larg_cm * CM, height=alt_cm * CM)
        p.insert_image(p.rect, stream=png.tobytes())
    doc.save(caminho, garbage=3, deflate=True)
    doc.close()


def caixa_da_tinta(pagina, dpi=30):
    """Retangulo da tinta (em fracao da pagina), lendo a pagina em baixa
    resolucao. Para o aviso de margem. None se a pagina estiver em branco."""
    pix = pagina.get_pixmap(dpi=dpi, colorspace=fitz.csGRAY)
    a = np.frombuffer(pix.samples, np.uint8).reshape(pix.h, pix.w)
    ys, xs = np.nonzero(a < 240)
    if len(xs) == 0:
        return None
    return (xs.min() / pix.w, ys.min() / pix.h, (xs.max() + 1) / pix.w, (ys.max() + 1) / pix.h)


def vaga(lado_esquerdo: bool, modo: str):
    """Area onde a pagina A5 pode ir, na folha A4 deitada.
    modo 'ajustar-papel': a A5 inteira menos os 3 mm da impressora;
    modo 'ajustar-margens': a A5 menos 1,3 cm fora/cima/baixo e 1,0 cm na costura."""
    meia = A4_DEITADA.width / 2
    x0 = 0 if lado_esquerdo else meia
    a5 = fitz.Rect(x0, 0, x0 + meia, A4_DEITADA.height)
    if modo == "ajustar-papel":
        fora, cost, cb = IMPRESSORA, 0.0, IMPRESSORA
    else:
        fora, cost, cb = FORA, COSTURA, FORA
    if lado_esquerdo:   # pagina par: costura a direita
        return a5, fitz.Rect(a5.x0 + fora * CM, a5.y0 + cb * CM, a5.x1 - cost * CM, a5.y1 - cb * CM)
    return a5, fitz.Rect(a5.x0 + cost * CM, a5.y0 + cb * CM, a5.x1 - fora * CM, a5.y1 - cb * CM)


def encaixe(src_rect, area, escala_extra=1.0):
    """'Ajustar' (cabe inteira, mantem proporcao) vezes a escala do lado,
    centrado na area."""
    f = min(area.width / src_rect.width, area.height / src_rect.height) * escala_extra
    w, h = src_rect.width * f, src_rect.height * f
    cx, cy = (area.x0 + area.x1) / 2, (area.y0 + area.y1) / 2
    return fitz.Rect(cx - w / 2, cy - h / 2, cx + w / 2, cy + h / 2), f


def impor(entrada, saida, por_caderno=20, modo="ajustar-papel", esc_esq=1.0, esc_dir=1.0,
          excecoes=None, guias=False, folhas=None):
    """excecoes: {indice_do_lado: (esc_esq, esc_dir)} = 'mudar uma folha so'.
    folhas: (de, ate) em lados, 1-based, para reimprimir so algumas."""
    excecoes = excecoes or {}
    src = fitz.open(entrada)
    out = fitz.open()
    avisos = []
    tinta = {}
    lados = montar_ordem(src.page_count, por_caderno)
    for n, lado in enumerate(lados, 1):
        if folhas and not (folhas[0] <= n <= folhas[1]):
            continue
        folha = out.new_page(width=A4_DEITADA.width, height=A4_DEITADA.height)
        ee, ed = excecoes.get(n, (esc_esq, esc_dir))
        for esquerda, idx, esc in ((True, lado.esquerda, ee), (False, lado.direita, ed)):
            a5, area = vaga(esquerda, modo)
            if idx is None:
                continue
            r, f = encaixe(src[idx].rect, area, esc)
            folha.show_pdf_page(r, src, idx)
            if idx not in tinta:
                tinta[idx] = caixa_da_tinta(src[idx])
            t = tinta[idx]
            if t is None:
                continue
            tx0, ty0 = r.x0 + t[0] * r.width, r.y0 + t[1] * r.height
            tx1, ty1 = r.x0 + t[2] * r.width, r.y0 + t[3] * r.height
            m = {"cima": ty0 / CM, "baixo": (a5.y1 - ty1) / CM}
            if esquerda:
                m["fora"], m["costura"] = (tx0 - a5.x0) / CM, (a5.x1 - tx1) / CM
            else:
                m["costura"], m["fora"] = (tx0 - a5.x0) / CM, (a5.x1 - tx1) / CM
            faltam = {k: round(v, 2) for k, v in m.items()
                      if v < (COSTURA if k == "costura" else FORA) - 1e-6}
            if faltam:
                avisos.append((idx + 1, n, round(f, 3), faltam))
        if guias:  # linha vermelha = margem minima (so na previa, nunca no arquivo de imprimir)
            for esq in (True, False):
                folha.draw_rect(vaga(esq, "ajustar-margens")[1], color=(1, 0, 0), dashes="[4] 0", width=1)
            folha.draw_line(fitz.Point(A4_DEITADA.width / 2, 0), fitz.Point(A4_DEITADA.width / 2, A4_DEITADA.height), color=(0, 0, 1), dashes="[2] 0", width=0.5)
    out.save(saida, garbage=3, deflate=True)
    out.close(); src.close()
    return lados, avisos


def png(pdf, n, destino, dpi=60):
    d = fitz.open(pdf)
    d[n].get_pixmap(dpi=dpi).save(destino)
    d.close()
