# ---------- JSON da pagina de escolhas ----------
saida = []
for i, p in enumerate(P):
    q = {"id": p["id"], "ordem": 127 + i, "status": "aberta", "quando": "20/10", "parte": "Impressão",
         "pergunta": p["pergunta"], "explicacao": p["explicacao"],
         "exemplos": [dict(e, src=IMG_PAG + e["src"], largo=True) for e in p["exemplos"]],
         "opcoes": [dict(o, imgs=[dict(m, src=IMG_PAG + m["src"]) for m in o["imgs"]]) for o in p["opcoes"]]}
    saida.append(q)
(D / "perguntas-impressao.json").write_text(json.dumps(saida, ensure_ascii=False, indent=1), encoding="utf-8")

# ---------- Markdown do relatorio ----------
L = ["# Seis perguntas da impressão, para o Samuel decidir",
     "",
     "Pesquisador, 20/10/2026. Pedido do Samuel: \"me explica melhor esses detalhes da impressão em um relatorio\".",
     "",
     "Lembrete do que já está decidido (14/10): o programa tem duas fases. Na edição, cada página ocupa a folha "
     "inteira. Só na hora de gerar o arquivo para imprimir se decide o formato: livreto (A4 deitada com duas páginas "
     "A5, já na ordem da dobra), A4 inteira em pé, ou as outras opções da janela do UPDF. Margens: 1,3 cm da beirada "
     "do papel em cima, embaixo e do lado de fora, e 1 cm na costura (a dobra).",
     "",
     "As seis perguntas abaixo são o que falta para começar. Cada uma tem os desenhos de cada opção, feitos pelo "
     "próprio programa de montar páginas que o Editor usa. A página dos desenhos é a página xi do Opus Majus, cortada rente ao texto. "
     "Quando uma opção diz **Sugestão**, é a minha sugestão, não uma decisão. A última opção de todas é sempre "
     "\"Ainda não entendi\".",
     ""]
for i, p in enumerate(P, 1):
    L += [f"## {i}. {p['pergunta']}", ""]
    for par in p["explicacao"].split("\n\n"):
        L += [par, ""]
    for e in p["exemplos"]:
        L += [f"![{e['legenda']}]({IMG_MD}{e['src']})", ""]
    letras = "ABCDE"
    for j, o in enumerate(p["opcoes"]):
        if o["id"] == "nao-entendi":
            L += [f"**Opção {letras[j]}: {o['nome']}.** {o['texto']}", ""]
            continue
        txt = o["texto"].replace("Sugestão:", "**Sugestão:**")
        L += [f"### Opção {letras[j]}: {o['nome']}", "", txt, ""]
        for m in o["imgs"]:
            L += [f"![{m['legenda']}]({IMG_MD}{m['src']})", ""]
L += ["## Como os desenhos foram feitos",
      "",
      "Com o PyMuPDF, a mesma biblioteca que o programa já usa para montar o PDF, num rascunho fora do programa "
      "(`desenhos_perguntas.py.txt`, ao lado deste relatório). A ordem das páginas nos cadernos vem do próprio "
      "programa (`core/cadernos.py`). As medidas escritas nos desenhos foram calculadas, não estimadas a olho. "
      "O Balzac (486 páginas) é o livro das suas fotos de 14/10; a conta dele também foi feita com a ordem do programa.",
      "",
      "O detalhe técnico de tudo está em `fase-impressao.pdf`, na mesma pasta."]
(D / "perguntas-para-o-samuel.md").write_text("\n".join(L) + "\n", encoding="utf-8")
print("ok", len(saida), "perguntas")
